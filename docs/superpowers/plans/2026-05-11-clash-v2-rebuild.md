#### 22.4 — 坐标映射 helper

新建 `src/clash/render/coords.ts`：
```ts
export const BOARD = { cols: 18, rows: 32 } as const
export function boardToPixel(pos: {x:number,y:number}, canvas: {width:number,height:number}): {x:number,y:number} {
  return { x: (pos.x / BOARD.cols) * canvas.width, y: (pos.y / BOARD.rows) * canvas.height }
}
export function pixelToBoard(px: {x:number,y:number}, canvas: {width:number,height:number}): {x:number,y:number} {
  return { x: (px.x / canvas.width) * BOARD.cols, y: (px.y / canvas.height) * BOARD.rows }
}
export function canvasSize(viewport: {w:number,h:number}): {w:number,h:number} {
  const aspectMatch = Math.min(viewport.w * 0.9 / BOARD.cols, viewport.h * 0.85 / BOARD.rows)
  return { w: aspectMatch * BOARD.cols, h: aspectMatch * BOARD.rows }
}
```

#### 22.6 — entityMap 骨架

```ts
appRef.current.ticker.add(() => {
  const state = stateRef.current
  if (!state) return
  // diff units
  for (const u of state.units) {
    if (!entityMapRef.current.has(u.id)) {
      const sprite = PIXI.Sprite.from(PIXI.Texture.WHITE) // placeholder
      sprite.tint = u.side === 'player' ? 0xb6ff3c : 0xff3c8c
      sprite.width = sprite.height = 20
      appRef.current!.stage.addChild(sprite)
      entityMapRef.current.set(u.id, sprite)
    }
    const sprite = entityMapRef.current.get(u.id)!
    const px = boardToPixel(u.pos, ...)
    sprite.x = px.x
    sprite.y = px.y
  }
  // cleanup vanished
  for (const [id, sprite] of entityMapRef.current) {
    if (!state.units.find(u => u.id === id)) {
      sprite.destroy()
      entityMapRef.current.delete(id)
    }
  }
})
```

```ts
export function useClashAssets(deckCardIds: CrCardId[]) {
  const [loaded, setLoaded] = useState(false)
  const [error, setError] = useState<Error | null>(null)
  useEffect(() => {
    const ac = new AbortController()
    PIXI.Assets.add({ alias: deckCardIds, src: deckCardIds.map(id => `/sprites/${id}.json`) })
    PIXI.Assets.load(deckCardIds, ac.signal).then(() => setLoaded(true)).catch(setError)
    return () => ac.abort()
  }, [deckCardIds.join(',')])
  return { loaded, error }
}
```

```sql
CREATE OR REPLACE FUNCTION cr_grant_starter_pack(p_user_id uuid)
RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE result json;
BEGIN
  -- idempotency check
  IF EXISTS(SELECT 1 FROM cr_user_cards WHERE user_id = p_user_id AND unlocked = true LIMIT 1) THEN
    SELECT json_build_object('granted', false, 'reason', 'already_initialized') INTO result;
    RETURN result;
  END IF;
  -- unlock 8 starter cards (already done by handle_new_user_clash trigger), just add bonus 200 gold
  UPDATE cr_account_currency SET gold = gold + 200 WHERE user_id = p_user_id;
  SELECT json_build_object('granted', true, 'gold_added', 200) INTO result;
  RETURN result;
END $$;

CREATE OR REPLACE FUNCTION dev_reset_user(p_user_id uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
BEGIN
  DELETE FROM cr_match_log WHERE user_id = p_user_id;
  DELETE FROM cr_chests WHERE user_id = p_user_id;
  UPDATE cr_user_cards SET unlocked = false, level = 1, shards = 0 WHERE user_id = p_user_id;
  -- re-unlock initial 8
  UPDATE cr_user_cards SET unlocked = true WHERE user_id = p_user_id
    AND card_id IN (SELECT id FROM cr_cards WHERE unlock_cost = 0);
  UPDATE cr_account_currency SET gold = 100, shards_total = 0 WHERE user_id = p_user_id;
END $$;
```

```ts
import type { MatchState, Action } from './types'

export interface TutorialScript {
  spawnGoblinAt: 30,  // tick (1s)
  inactivityHintTicks: [1800, 2700, 3600],  // 60s, 90s, 120s
  endOnPrincessDestroyed: 'enemy_left',
}

export function tutorialDecide(state: MatchState, script: TutorialScript): Action | null {
  // Only spawn 1 goblin at enemy_left bridge, hold fire after player kills
  // No king destruct - player must destroy princess themselves
  if (state.tick === script.spawnGoblinAt && state.enemy.units.length === 0) {
    return { kind: 'deploy', side: 'enemy', handIndex: 0, pos: { x: 3, y: 22 } }
  }
  return null
}

export function tutorialIsComplete(state: MatchState): boolean {
  const enemyLeft = state.towers.find(t => t.id === 'enemy_left')
  return enemyLeft ? enemyLeft.hp <= 0 : false
}
```
